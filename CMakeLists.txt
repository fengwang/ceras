cmake_minimum_required(VERSION 3.21)
project(ceras LANGUAGES CXX)
include(CTest)
option(CERAS_BUILD_TESTS "Build maintained regression tests" ON)
option(CERAS_ENABLE_CBLAS "Use CBLAS" OFF)
option(CERAS_ENABLE_CUDA "Use CUDA/cuBLAS" OFF)
option(CERAS_ENABLE_ASAN "Address sanitizer" OFF)
option(CERAS_ENABLE_UBSAN "Undefined behavior sanitizer" OFF)
option(CERAS_ENABLE_TSAN "Thread sanitizer" OFF)
set(CERAS_MNIST_DIR "" CACHE PATH "Enable the optional seeded MNIST gate with this local dataset")
option(CERAS_STRICT_FP "Preserve floating point semantics" ON)
find_package(Threads REQUIRED)
add_library(ceras INTERFACE)
add_library(ceras::ceras ALIAS ceras)
target_include_directories(ceras INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ceras INTERFACE cxx_std_20)
target_link_libraries(ceras INTERFACE Threads::Threads)
if(CERAS_ENABLE_CBLAS)
  find_package(BLAS REQUIRED)
  include(CheckFunctionExists)
  set(CMAKE_REQUIRED_LIBRARIES ${BLAS_LIBRARIES})
  check_function_exists(cblas_dgemm CERAS_HAS_CBLAS)
  if(NOT CERAS_HAS_CBLAS)
    message(FATAL_ERROR "Selected BLAS does not supply cblas_dgemm")
  endif()
  target_compile_definitions(ceras INTERFACE CBLAS)
  target_link_libraries(ceras INTERFACE BLAS::BLAS)
endif()
if(CERAS_ENABLE_CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(ceras INTERFACE CUDA)
  target_link_libraries(ceras INTERFACE CUDA::cudart CUDA::cublas)
endif()
if(CERAS_ENABLE_TSAN AND (CERAS_ENABLE_ASAN OR CERAS_ENABLE_UBSAN))
  message(FATAL_ERROR "Run TSan in a separate build")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  if(CERAS_STRICT_FP)
    target_compile_options(ceras INTERFACE -fno-fast-math)
  endif()
  foreach(pair "ASAN;address" "UBSAN;undefined" "TSAN;thread")
    list(GET pair 0 opt)
    list(GET pair 1 flag)
    if(CERAS_ENABLE_${opt})
      target_compile_options(ceras INTERFACE -fsanitize=${flag} -fno-omit-frame-pointer)
      target_link_options(ceras INTERFACE -fsanitize=${flag})
    endif()
  endforeach()
endif()
add_library(ceras_image STATIC src/image.cpp)
add_library(ceras::image ALIAS ceras_image)
target_link_libraries(ceras_image PUBLIC ceras)
if(CERAS_BUILD_TESTS)
  if(CERAS_ENABLE_CUDA)
    add_executable(ceras_cuda_concurrency test/backend_concurrency.cc)
    target_link_libraries(ceras_cuda_concurrency PRIVATE ceras)
    add_test(NAME cuda_concurrency COMMAND ceras_cuda_concurrency)
    set_tests_properties(cuda_concurrency PROPERTIES LABELS "cuda;concurrency" TIMEOUT 120 SKIP_RETURN_CODE 77)
  endif()
  add_executable(ceras_headers test/headers.cc)
  target_link_libraries(ceras_headers PRIVATE ceras)
  add_test(NAME headers COMMAND ceras_headers)
  add_executable(ceras_verification test/verification.cc)
  target_link_libraries(ceras_verification PRIVATE ceras)
  foreach(case gradients boundaries contexts recurrences optimizer_state normalization training allocation restore ranges parsers)
    add_test(NAME ${case} COMMAND ceras_verification ${case})
    set_tests_properties(${case} PROPERTIES TIMEOUT 60)
  endforeach()
  set_tests_properties(ranges contexts PROPERTIES LABELS concurrency)
  add_custom_target(fuzz-smoke COMMAND $<TARGET_FILE:ceras_verification> parsers DEPENDS ceras_verification)
  add_executable(ceras_api test/api.cc)
  target_link_libraries(ceras_api PRIVATE ceras)
  add_test(NAME api COMMAND ceras_api)
  add_executable(ceras_multi_tu test/multi_tu.cc test/multi_tu_other.cc)
  target_link_libraries(ceras_multi_tu PRIVATE ceras)
  add_test(NAME multi_tu COMMAND ceras_multi_tu)
  add_executable(ceras_regression test/regression.cc test/image_other.cc)
  target_link_libraries(ceras_regression PRIVATE ceras_image)
  foreach(case gemm storage parse image hdr idx parallel graph softmax mean elu gelu maximum lifetime optimizers adadelta crossentropy)
    add_test(NAME ${case} COMMAND ceras_regression ${case})
    set_tests_properties(${case} PROPERTIES TIMEOUT 60)
  endforeach()
  set_tests_properties(parallel PROPERTIES LABELS concurrency)
endif()

add_executable(ceras_benchmark EXCLUDE_FROM_ALL test/benchmark.cc)
target_link_libraries(ceras_benchmark PRIVATE ceras)
add_custom_target(benchmarks COMMAND $<TARGET_FILE:ceras_benchmark> DEPENDS ceras_benchmark)
if(CERAS_BUILD_TESTS AND CERAS_MNIST_DIR)
  add_executable(ceras_mnist_gate test/mnist_gate.cc)
  target_link_libraries(ceras_mnist_gate PRIVATE ceras)
  add_test(NAME mnist_integration COMMAND ceras_mnist_gate "${CERAS_MNIST_DIR}")
  set_tests_properties(mnist_integration PROPERTIES TIMEOUT 300 LABELS integration)
endif()

add_executable(ceras_modern_features EXCLUDE_FROM_ALL test/modern_features.cc)
target_link_libraries(ceras_modern_features PRIVATE ceras)
